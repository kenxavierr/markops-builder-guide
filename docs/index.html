SQL Fundamentals

SQL helps you filter, segment, and prepare data before sending or creating target audiences.

SELECT from a Data Extension
SELECT
  SubscriberKey,
  EmailAddress,
  FirstName,
  LastName
FROM ent.CustomerMaster
WHERE filters
SELECT
  SubscriberKey,
  EmailAddress,
  FirstName
FROM ent.CustomerMaster
WHERE IsActive = 1
  AND Country = 'Malaysia'

Utility Queries

These are helpful when checking statuses, reviewing data volume, and removing duplicate records before you build a final audience.

Query 1 · Status count check
SELECT
  a.Status,
  COUNT(*) AS Cnt
FROM DEName a
WHERE a.Status IN ('Active', 'Cancelled')
GROUP BY a.Status
Query 2 · Campaign status volume
SELECT
  a.CampaignCode__c,
  a.Status,
  COUNT(*) AS Cnt
FROM ent.Salesforce_Sample a
WHERE a.CampaignCode__c = '000000123'
  AND a.Status IN ('Active', 'Cancelled')
GROUP BY
  a.CampaignCode__c,
  a.Status
Query 3 · Total record volume by status
SELECT
  a.Status,
  COUNT(*) AS TotalVolume
FROM ent.Salesforce_Sample a
GROUP BY a.Status
ORDER BY TotalVolume DESC
Query 4 · Volume by campaign code
SELECT
  a.CampaignCode__c,
  COUNT(*) AS TotalRecords
FROM ent.Salesforce_Sample a
GROUP BY a.CampaignCode__c
ORDER BY TotalRecords DESC
Query 5 · Deduplicate records
SELECT
  SubscriberKey,
  EmailAddress,
  FirstName,
  Country
FROM (
  SELECT
    SubscriberKey,
    EmailAddress,
    FirstName,
    Country,
    ROW_NUMBER() OVER (
      PARTITION BY SubscriberKey
      ORDER BY CreatedDate DESC
    ) AS rn
  FROM ent.CustomerMaster
) x
WHERE x.rn = 1
